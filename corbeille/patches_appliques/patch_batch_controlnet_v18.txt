# ── REMBG PIL INLINE ──────────────────────────────────────────────────────────
def prepare_prev_for_i2i(img_path: str) -> str:
    """
    Isole le personnage sur fond blanc pur avant VAEEncode img2img.
    Supprime le bruit de fond accumulé sans modèle externe.
    Retourne le chemin d'un fichier temporaire dans COMFY_INPUT.
    """
    if not HAS_PIL:
        return img_path
    try:
        img = Image.open(img_path).convert("RGBA")
        arr = __import__("numpy").array(img)
        # Masque fond clair : pixel "blanc" si R+G+B > seuil*3
        r, g, b = arr[:,:,0], arr[:,:,1], arr[:,:,2]
        is_bg = (r.astype(int) + g.astype(int) + b.astype(int)) > (REMBG_THRESHOLD * 3)
        # Fond → blanc pur, personnage conservé
        result = arr.copy()
        result[is_bg] = [255, 255, 255, 255]
        out_img = Image.fromarray(result).convert("RGB")
        # Sauvegarde temporaire
        tmp_path = os.path.join(COMFY_INPUT, "_rembg_tmp.png")
        out_img.save(tmp_path)
        return tmp_path
    except Exception as e:
        print(f"[WARN] prepare_prev_for_i2i: {e} — fallback original")
        return img_path
