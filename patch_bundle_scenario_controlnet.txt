    "one piece": (

    ),
}


def _get_universe_constraint(universe: str) -> str:
    """Retourne la contrainte système pour l'univers donné."""
    u = universe.lower()
    for key, constraint in UNIVERSE_SYSTEM_CONSTRAINTS.items():
        if key in u:
            return constraint
    return ""
